// React imports for component creation and state management
import React, { useState } from "react";

// React Native UI components
import {
  SafeAreaView,
  Text,
  TextInput,
  Button,
  ScrollView,
  View,
} from "react-native";

// Function that calls the Python backend to generate a truth table
import { getTruthTable } from "./src/api/logic";

// Main application component
export default function App() {
  // Stores the user-entered logical formula
  const [formula, setFormula] = useState("p -> q");

  // Stores output returned from the backend (JSON as string for now)
  const [output, setOutput] = useState<string>("");

  // Called when the user presses the "Generate Truth Table" button
  async function run() {
    // Give user immediate feedback
    setOutput("Loading...");

    try {
      // Send formula to backend and wait for response
      const data = await getTruthTable(formula);

      // Convert JSON into readable text
      setOutput(JSON.stringify(data, null, 2));
    } catch (e: any) {
      // Handle errors cleanly
      setOutput(`Error: ${e?.message ?? String(e)}`);
    }
  }

  return (
    <SafeAreaView style={{ flex: 1, padding: 16 }}>
      {/* App title */}
      <Text style={{ fontSize: 20, marginBottom: 8 }}>
        Logic Generator
      </Text>

      {/* Input label */}
      <Text style={{ marginBottom: 6 }}>
        Enter a formula:
      </Text>

      {/* Text input for logical formula */}
      <TextInput
        value={formula}
        onChangeText={setFormula}
        placeholder="e.g. (p & q) -> r"
        autoCapitalize="none"
        autoCorrect={false}
        style={{
          borderWidth: 1,
          borderColor: "#ccc",
          padding: 12,
          borderRadius: 8,
          marginBottom: 12,
        }}
      />

      {/* Button to trigger backend request */}
      <Button title="Generate Truth Table" onPress={run} />

      {/* Scrollable output area */}
      <ScrollView style={{ marginTop: 12 }}>
        <View>
          <Text
            selectable
            style={{ fontFamily: "Courier", fontSize: 12 }}
          >
            {output}
          </Text>
        </View>
      </ScrollView>
    </SafeAreaView>
  );
}
